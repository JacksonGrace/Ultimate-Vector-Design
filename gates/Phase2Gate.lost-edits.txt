    /** Two geo pieces + an OPTIMIZED bridge whose entry is pitched AND
     *  turning — the march carries a pitch hump, the case whose fresh
     *  sub-march cannot reproduce the old sub-arc. */
    static Coaster pcmFixture(BridgeSection[] out) {
        Coaster c = new Coaster();
        TrackPiece p0 = new TrackPiece(0, 21, 0, 0);
        List<List<BezierBaker.Node>> l0 = new ArrayList<>();
        l0.add(nodes(new double[][]{ {0, 15}, {1, 15} }));
        l0.add(nodes(new double[][]{ {0, 25}, {1, 25} }));
        l0.add(nodes(new double[][]{ {0, 0}, {1, 0} }));
        p0.sections.add(new GeoSection(l0, 26));
        c.pieces.add(p0);
        TrackPiece p1 = new TrackPiece(60, 15, 20, 0);
        List<List<BezierBaker.Node>> l1 = new ArrayList<>();
        for (int i = 0; i < 3; i++) {
            l1.add(nodes(new double[][]{ {0, 0}, {1, 0} }));
        }
        p1.sections.add(new GeoSection(l1, 26));
        c.pieces.add(p1);
        TrackCompiler.Result pre = TrackCompiler.compile(c, 0.01);
        BridgeSection bs = BridgeSection.solve(
            pre.spans().get(0).exit(), p1.entryPose(), 26);
        p0.sections.add(bs);
        TrackCompiler.compile(c, 0.01);        // capture seam dynamics
        c.optimizeBridge(bs, 1, 30, true);     // march-rendering
        TrackCompiler.compile(c, 0.01);
        out[0] = bs;
        return c;
    }

    static void partialConvertMarch() {
        for (boolean left : new boolean[]{ false, true }) {
            String dir = left ? "LEFT" : "RIGHT";
            BridgeSection[] bh = new BridgeSection[1];
            Coaster c = pcmFixture(bh);
            TrackCompiler.Result before = TrackCompiler.compile(c, 0.01);
            check(!c.autoReoptimizeBridges(before),
                "pcm " + dir + ": optimized fixture starts settled");
            check(c.partialConvertBridge(0, 1, 0.5, left, true, 0.01),
                "pcm " + dir + ": applies");
            BridgeSection sb = left
                ? (BridgeSection) c.pieces.get(0).sections.get(2) : bh[0];
            TrackPiece tgt = c.pieces.get(1);
            double[] cutAt = { tgt.x, tgt.y, tgt.z };
            // the app's rebuild loop: compile -> auto pass -> settle
            TrackCompiler.Result r1 = TrackCompiler.compile(c, 0.01);
            check(c.autoReoptimizeBridges(r1),
                "pcm " + dir + ": stale signature fires the auto pass once");
            double nudge = new Vec3(tgt.x - cutAt[0], tgt.y - cutAt[1],
                tgt.z - cutAt[2]).length();
            TrackCompiler.Result r2 = TrackCompiler.compile(c, 0.01);
            check(!c.autoReoptimizeBridges(r2),
                "pcm " + dir + ": ONE recompile settles");
            check(nudge > 1e-6 && nudge < 0.5, String.format(
                "pcm " + dir + ": re-place nudge bounded (%.3f m)", nudge));
            // landing proxy: the placement now sits ON the fresh march,
            // so a same-mode re-optimize at the same length moves
            // ~nothing — exactly marchSpine's landing condition (march
            // renders; pre-fix this moved by the 0.09-0.17 m miss)
            double[] pp = { tgt.x, tgt.y, tgt.z };
            c.optimizeBridge(sb, 1, sb.lastOptDist,
                sb.optMode == BridgeSection.OPT_POS_YAW);
            double moved = new Vec3(tgt.x - pp[0], tgt.y - pp[1],
                tgt.z - pp[2]).length();
            check(moved < 0.02, String.format("pcm " + dir
                + ": placement sits ON the march — landing gate holds"
                + " (re-optimize moves %.5f m)", moved));
            // the whole track stays where the user cut it (fit
            // residual + the bounded nudge)
            TrackCompiler.Result r3 = TrackCompiler.compile(c, 0.01);
            double[] dev = shapeDev(before.samples(), r3.samples());
            check(dev[0] < 0.6 && dev[1] < 0.6, String.format(
                "pcm " + dir + ": whole track holds (interior %.3f m,"
                + " end %.3f m)", dev[0], dev[1]));

        // record-only null: the auto pass must NOT move the cut

            "pcm septic: never-optimized bridge stays record-only —"
            + " nothing moves");

    /** Position INTERPOLATED at arc s (atS snaps to the nearest sample
     *  — up to half a sample of along-track offset reads as fake
     *  deviation when two compiles' grids don't line up). */
    static Vec3 posAtS(List<FVDIntegrator.Sample> l, double s) {
        int lo = 0, hi = l.size() - 1;
        while (hi - lo > 1) {
            int mid = (lo + hi) >>> 1;
            if (l.get(mid).s() <= s) lo = mid; else hi = mid;
        }
        FVDIntegrator.Sample a = l.get(lo), b = l.get(hi);
        double ds = b.s() - a.s();
        double u = ds < 1e-12 ? 0
            : Math.max(0, Math.min(1, (s - a.s()) / ds));
        return a.pos().add(b.pos().sub(a.pos()).scale(u));
    }
